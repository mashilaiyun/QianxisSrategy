题目描述

模拟一套简化的序列化传输方式，请实现下面的数据编码与解码过程

编码前数据格式为 [位置,类型,值]，多个数据的时候用逗号分隔，位置仅支持数字，不考虑重复等场景；类型仅支持：Integer / String / Compose（Compose的数据类型表示该存储的数据也需要编码）。

编码后数据参考图示，数据区的格式是：位置#类型#长度#数据，类型存储需要编码，Integer->0；String->1；Compose->2，长度是指数据的字符长度；数据仅允许数字、大小写字母、空格。

输入的编码字符长度不能超过1000，一个数据的格式错误，则解析剩下数据，其他错误输出ENCODE_ERROR。

输入的解码字符不能超过1000，数据区异常则跳过继续解析剩余数据区，其他异常输出DECODE_ERROR。

 

输入描述

输入有两行：

第一行是命令，1表示编码，2表示解码, 第二行输入待编码、解码的字符

数据最多嵌套10层，[1,Compose,[1,String,Second]] 为2层嵌套

输出描述

如果输入要求是编码，则输出编码结果；如果输入要求是解码，则输出解码结果；当异常时输出对应的错误字符

用例1

输入

xxxxxxxxxx
1
[1,String,I am Mary],[2,Integer,23],[3,Long,1000000],[4,Compose,[1,String,I am Kitty],[2,Integer,44]]

输出

xxxxxxxxxx
1#1#9#I am Mary2#0#2#234#2#25#1#1#10#I am Kitty2#0#2#44

说明

由于Long型未不支持类型，所以数据[3,Long,1000000]自动被过滤

用例2

输入

xxxxxxxxxx
2
1#1#9#I am Mary2#0#2#233#0#3#8794#2#25#1#1#10#I am Kitty2#0#2#44

输出

xxxxxxxxxx
[1,String,I am Mary],[2,Integer,23],[3,Integer,879],[4,Compose,[1,String,I am Kitty],[2,Integer,44]]

题解

思路：字符串处理 + 递归

  1. 此题考察字符串处理，以及考察处理嵌套类型时递归的应用。
  2. 编码和解码的情况分别定义一个函数去进行处理。在编码和解码两种情况下对于嵌套类型的处理都可以使用递归去进行处理。
  3. 编码和解码具体逻辑可以参照下面对应代码实现。

vector<string> splitTopLevelItem(string& input) {
    vector<string> res;
    int depth = 0;
    int start = 0;
    int n = input.size();
    for (int i = 0; i < n; i++) {
        char c = input[i];
        if (c == '[') {
            if (depth == 0) {
                start = i;
            }
            depth += 1;
        } else if (c == ']') {
            depth -= 1;
            if (depth == 0) {
                string tmp = input.substr(start, i - start + 1);
                res.push_back(tmp);
            }
        }
        // 出现括号乱序问题 右括号大于左括号数量
        if (depth < 0) {
            flag = true;
            break;
        }
    }
​
    // 括号左右对数不匹配
    if (depth != 0) {
        flag = true;
    }
    return res;
}

// 通用 split 函数
vector<string> split(const string& str, const string& delimiter) {
    vector<string> result;
    size_t start = 0;
    size_t end = str.find(delimiter);
    while (end != string::npos) {
        result.push_back(str.substr(start, end - start));
        start = end + delimiter.length();
        end = str.find(delimiter, start);
    }
    // 添加最后一个部分
    result.push_back(str.substr(start));
    return result;
}

// 解码
vector<Item> decode(string& input) {
    vector<Item> res;
    vector<string> parts = split(input, "#");
    int m = parts.size();
    string pos = "";
    string type = "";
    string value = "";
    int count = -1;
    for (int i = 0; i < m; i++) {
        // 按顺序赋值
        if (pos == "") {
            pos = parts[i];
        } else if (type == "") {
            type = parts[i];
        } else if (count == -1) {
            string countStr = parts[i];
            count = stoi(countStr);
        } else {
            // 值类型
            if (type == "1" || type == "0") {
                value = parts[i];
                // 两种情况:
                // 1. 最后一个数据,长度应该等于count
                // 2. 不是最后一个数组,长度应该大于count , 由 内容 + 下一个数据的位置编号
                if ((i != m -1 && value.size() <= count) || (i == (m - 1) && value.size() != count)) {
                    flag = true;
                    break;
                }
                string remainngStr = value.substr(count);
                value = value.substr(0, count);
                res.push_back({pos, type, count, value});
                parts[i] = remainngStr;
                // 因为最后一个元素的内容和下一个元素的id会在一起 
                i--;
            } else {
                int tmpCount = 0;
                int j = i;  
                string childrenInput = "";
                // 通过长度找出复合类型的内容编码字符串
                childrenInput = getChildrenInput(parts, j, count);
                if (flag) {
                    break;
                }
                res.push_back({pos, type, count, ""});
                vector<Item> childrenItem = decode(childrenInput);
                res.back().children = childrenItem;
                // 因为最后一个元素的内容和下一个元素的id会在一起 
                i = j-1;
            }
​
            // 重置
            pos = "";
            type = "";
            count = -1;
            value = "";
        }
    }
​
    // 数据不完整
    if (pos != "" || type != "" || count != -1 || value != "") {
        flag = true;
    }
​
    return res;
}

function decode(input) {
  const res = [];
  const parts = split(input, '#');
  const m = parts.length;
  let pos = '';
  let type = '';
  let value = '';
  let count = -1;
  
  for (let i = 0; i < m; i++) {
    // 按顺序赋值
    if (pos === '') {
      pos = parts[i];
    } else if (type === '') {
      type = parts[i];
    } else if (count === -1) {
      count = parseInt(parts[i], 10);
    } else {
      // 根据不同类型 不同处理方式
      if (type === '1' || type === '0') {
        value = parts[i];
        // 两种情况:
        // 1. 最后一个数据,长度应该等于count
        // 2. 不是最后一个数组,长度应该大于count , 由 内容 + 下一个数据的位置编号
        if ((i !== m - 1 && value.length <= count) || (i === m - 1 && value.length !== count)) {
          flag = true;
          break;
        }
        const remainngStr = value.substring(count);
        value = value.substring(0, count);
        res.push(new Item(pos, type, count, value));
        parts[i] = remainngStr;
        // 因为最后一个元素的内容和下一个元素的id会在一起 
        i--;
      } else {
        const j = [i];
        const childrenInput = getChildrenInput(parts, j, count);
        if (flag) break;
        const newItem = new Item(pos, type, count, '');
        newItem.children = decode(childrenInput);
        res.push(newItem);
        // 因为最后一个元素的内容和下一个元素的id会在一起 
        i = j[0] - 1;
      }
      // 重置
      pos = '';
      type = '';
      count = -1;
      value = '';
    }
  }
  if (pos !== '' || type !== '' || count !== -1 || value !== '') {
    flag = true;  // 解码数据不完整
  }
  return res;
}

// splitTopLevelItem 按中括号层级分割顶层子项
func splitTopLevelItem(input string) []string {
    res := []string{}
    depth := 0
    start := 0
    for i, c := range input {
        if c == '[' {
            if depth == 0 {
                start = i
            }
            depth++
        } else if c == ']' {
            depth--
            if depth == 0 {
                res = append(res, input[start:i+1])
            }
        }
        // 括号乱序
        if depth < 0 {
            flag = true
            break
        }
    }
    // 左右括号数量不一致
    if depth != 0 {
        flag = true
    }
    return res
}

// encode 编码函数，将字符串转为Item切片
func encode(input string) []Item {
    res := []Item{}
    if input == "" {
        return res
    }
    // 判断格式
    if input[0] != '[' || input[len(input)-1] != ']' {
        flag = true
        return res
    }
​
    ans := splitTopLevelItem(input)
    if flag {
        return res
    }
​
    for _, itemStr := range ans {
        // 去除首尾中括号
        itemStr = itemStr[1 : len(itemStr)-1]
​
        currentAns := []string{}
        tmp := ""
        j := 0
        for j < len(itemStr) {
            c := itemStr[j]
            if c != ',' {
                tmp += string(c)
            } else {
                currentAns = append(currentAns, tmp)
                tmp = ""
                if len(currentAns) == 2 {
                    break
                }
            }
            j++
        }
 
        typ := currentAns[len(currentAns)-1]
        // 根据不同类型不同处理方式
        if typ == "Integer" || typ == "String" {
            value := itemStr[j+1:]
            count := len(value)
            typeStr := "0"
            if currentAns[1] == "String" {
                typeStr = "1"
            }
            res = append(res, Item{pos: currentAns[0], typ: typeStr, count: count, value: value})
        } else if typ == "Compose" {
            childrenItemStr := itemStr[j+1:]
            // 递归处理复合类型
            children := encode(childrenItemStr)
            // 计算复合类型长度
            count := 0
            for _, item := range children {
                count += item.count + typeLength[item.typ]
            }
            composeItem := Item{pos: currentAns[0], typ: "2", count: count, value: ""}
            composeItem.children = children
            res = append(res, composeItem)
        }
    }
    return res
}

// decode 解码字符串为Item列表
func decode(input string) []Item {
    res := []Item{}
    parts := split(input, "#")
    m := len(parts)
    pos := ""
    typ := ""
    value := ""
    count := -1
    for i := 0; i < m; i++ {
        if pos == "" {
            pos = parts[i]
        } else if typ == "" {
            typ = parts[i]
        } else if count == -1 {
            var err error
            count, err = strconv.Atoi(parts[i])
            if err != nil {
                flag = true
                break
            }
        } else {
            if typ == "1" || typ == "0" {
                value = parts[i]
                // 两种情况:
                // 1. 最后一个数据,长度应该等于count
                // 2. 不是最后一个数组,长度应该大于count , 由 内容 + 下一个数据的位置编号
                if (i != m-1 && len(value) <= count) || (i == m-1 && len(value) != count) {
                    flag = true
                    break
                }
                remainingStr := ""
                if len(value) > count {
                    remainingStr = value[count:]
                }
                value = value[:count]
                res = append(res, Item{pos: pos, typ: typ, count: count, value: value})
                parts[i] = remainingStr
                // 因为一个元素的内容和下一个元素的id会在一起 
                i--
            } else {
                j := i
                childrenInput := getChildrenInput(parts, &j, count)
                if flag {
                    break
                }
                newItem := Item{pos: pos, typ: typ, count: count, value: ""}
                newItem.children = decode(childrenInput)
                res = append(res, newItem)
                // 因为最后一个元素的内容和下一个元素的id会在一起
                i = j - 1
            }
            pos = ""
            typ = ""
            count = -1
            value = ""
        }
    }
    // 不完整
    if pos != "" || typ != "" || count != -1 || value != "" {
        flag = true
    }
    return res
}
