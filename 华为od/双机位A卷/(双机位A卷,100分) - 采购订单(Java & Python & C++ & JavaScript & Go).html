题目描述

在一个采购系统中，采购申请(PR)需要经过审批后才能生成采购订单(PO)。每个PR包含商品的单价(假设相同商品的单价一定是一样的)及数量信息。系统要求对商品进行分类处理:单价高于100元的商品需要单独处理，单价低于或等于100元的相同商品可以合并到同一采购订单PO中。针对单价低于100的小额订单，如果量大可以打折购买。

具体规则如下：

如果PR状态为"审批通过"，则将其商品加入到PO中。如果PR的状态为"审批拒绝"或"待审批"，则忽略改PR。

对于单价高于100元的商品，每个商品单独生成一条PO记录。对于单价低于100元的商品，将相同商品的数量合并到一条PO记录中。

如果商品单价<100且商品数量>=100，则单价打9折。

输入描述

第一行包含整数N，表示PR的数量。

接下来N行，每行包含五个用空格分割的整数，按顺序表示：商品ID,数量，单价，PR状态(0表示审批通过，1表示审批拒绝，2表示待审批)

输出描述

输出若干行，每行表示一条PO记录，按以下格式输出：

对于单价高于100元的商品：商品ID 数量 单价

对于单价低于或等于100元的商品： 商品ID 总数量 打折后的单价(向上取整)

输出的PO记录按商品ID升序升序排列，相同商品按照数量降序排列

补充

  • 2 <= n <= 1000
  • 1 <= 商品价格 <= 200
  • 1 <= 商品数量 <= 1000
  • 2 <= 商品编号 <= 1000

用例1

输入

xxxxxxxxxx
2
1 200 90 0
2 30 101 0

输出

xxxxxxxxxx
1 200 81
2 30 101

说明

商品1的原始单价为90，审批通过，生成一条PO，满足打折条件，打折后单价为81。

商品2的单价为101，审批通过，生成一条PO

用例2

输入

xxxxxxxxxx
3
1 10 90 0
1 5 90 0
2 8 120 0

输出

xxxxxxxxxx
1 15 90
2 8 120

说明

PR1和PR2均为商品1，单价90，审批通过，单价低于100元，合并数量为150.

PR3为商品2，单价120元，审批通过，单价高于100元，单独生成一条PO记录。

用例3

输入

xxxxxxxxxx
4
1 5 80 0
2 3 120 0
3 2 90 1
4 10 150 2

输出

xxxxxxxxxx
1 5 80
2 3 120

说明

PR1：商品1，单价80元，审批通过，单价低于100元，合并到PO中。PR2：商品2，单价120元，审批通过，单价高于100元，单独生成一条PO记录。PR3：审批拒绝， 忽略。PR4待审批忽略。

 

题解

思路：模拟

  1. 循环处理输入的PR，两种策略处理审批通过不同单价产品：

    • 单价> 100的PR: 直接生成一条最终PO记录。
    • 单价 <= 100的PR：可以使用哈希表累计对应商品总数目

  2. 处理1中哈希表每个单价低于100的商品，为它们生成PO，如果该商品累计总数目大于等于100，折扣之后价格为ceil(price * 0.9)向上取整，添加一条新的PO记录。

  3. 对于1 2 得到的最终PO记录，进行自定义排序.排序规则为商品ID升序升序排列，相同商品按照数量降序排列

  4. 按顺序和题目要求规则输出PO记录。

#include <iostream>
#include <vector>
#include <map>
#include <cmath>
#include <algorithm>
using namespace std;
​
struct Product {
    int id;
    int price;
    int num;
    int discountPrice;
    Product(int id, int price, int num)
        : id(id), price(price), num(num), discountPrice(price) {}
};
​
int main() {
    int n;
    cin >> n;
​
    // 存储低价商品（<=100），key为id，value为{数量, 单价}
    map<int, pair<int, int>> order;
​
    vector<Product> res;
​
    for (int i = 0; i < n; i++) {
        int id, num, price, status;
        cin >> id >> num >> price >> status;
​
        if (status != 0) continue; // 只处理审批通过
​
        if (price > 100) {
            // 高价商品，单独成PO
            res.emplace_back(id, price, num);
        } else {
            // 低价商品，合并数量
            order[id].first += num;
            order[id].second = price;
        }
    }
​
    // 处理低价商品合并结果
    for (auto &p : order) {
        int id = p.first;
        int num = p.second.first;
        int price = p.second.second;
​
        Product tmp(id, price, num);
        if (num >= 100) {
            tmp.discountPrice = static_cast<int>(ceil(price * 0.9));
        }
        res.push_back(tmp);
    }
​
    // 按id升序，数量降序
    sort(res.begin(), res.end(), [](const Product &a, const Product &b) {
        if (a.id == b.id) return a.num > b.num;
        return a.id < b.id;
    });
​
    // 输出
    for (auto &p : res) {
        cout << p.id << " " << p.num << " " << p.discountPrice << "\n";
    }
​
    return 0;
}

import java.util.*;
import java.io.*;
import java.math.*;
​
public class Main {
    static class Product {
        int id;
        int price;
        int num;
        int discountPrice;
        Product(int id, int price, int num) {
            this.id = id;
            this.price = price;
            this.num = num;
            this.discountPrice = price;
        }
    }
​
    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine().trim());
        Map<Integer, int[]> order = new HashMap<>(); // 低价商品汇总
        List<Product> res = new ArrayList<>();
​
        for (int i = 0; i < n; i++) {
            String[] parts = br.readLine().trim().split(" ");
            int id = Integer.parseInt(parts[0]);
            int num = Integer.parseInt(parts[1]);
            int price = Integer.parseInt(parts[2]);
            int status = Integer.parseInt(parts[3]);
​
            if (status != 0) continue; // 只处理审批通过
​
            if (price > 100) {
                res.add(new Product(id, price, num)); // 高价商品单独成PO
            } else {
                order.putIfAbsent(id, new int[]{0, price});
                order.get(id)[0] += num; // 累加数量
            }
        }
​
        // 处理低价商品
        for (Map.Entry<Integer, int[]> entry : order.entrySet()) {
            int id = entry.getKey();
            int num = entry.getValue()[0];
            int price = entry.getValue()[1];
            Product tmp = new Product(id, price, num);
            if (num >= 100) {
                tmp.discountPrice = (int) Math.ceil(price * 0.9);
            }
            res.add(tmp);
        }
​
        // 按id升序，数量降序
        res.sort((a, b) -> a.id == b.id ? b.num - a.num : a.id - b.id);
​
        // 输出
        StringBuilder sb = new StringBuilder();
        for (Product p : res) {
            sb.append(p.id).append(" ").append(p.num).append(" ").append(p.discountPrice).append("\n");
        }
        System.out.print(sb.toString());
    }
}

const readline = require('readline');

const rl = readline.createInterface({
  input: process.stdin,
  output: process.stdout

class Product {

    this.id = id;

    this.num = num;
    this.discountPrice = price;
  }
}
​
let n = 0;
let lines = [];
const res = [];
const order = new Map(); // 存储低价商品（<=100）的合并记录 id -> [数量, 单价]
​
rl.on('line', (line) => {
  line = line.trim();
  if (!line) return;
  lines.push(line);
});
​
rl.on('close', () => {
  n = parseInt(lines[0]); // 第一行是数量
  for (let i = 1; i <= n; i++) {
    const [id, num, price, status] = lines[i].split(' ').map(Number);
    if (status !== 0) continue; // 忽略未审批通过的
​
    if (price > 100) {
      // 高价商品：每个单独成PO
      res.push(new Product(id, price, num));
    } else {
      // 低价商品：同id合并
      if (!order.has(id)) {
        order.set(id, [0, price]);
      }
      const [count, p] = order.get(id);
      order.set(id, [count + num, p]);
    }
  }
​
  // 处理低价商品合并结果
  for (const [id, [num, price]] of order.entries()) {
    const tmp = new Product(id, price, num);
    if (num >= 100) {
      tmp.discountPrice = Math.ceil(price * 0.9);
    }
    res.push(tmp);
  }
​
  // 按 id 升序，数量降序
  res.sort((a, b) => {
    if (a.id === b.id) return b.num - a.num;
    return a.id - b.id;
  });
​
  // 输出结果
  for (const p of res) {
    console.log(`${p.id} ${p.num} ${p.discountPrice}`);
  }
});

package main
​
import (
    "bufio"
    "fmt"
    "math"
    "os"
    "sort"
)
​
type Product struct {
    id            int
    price         int
    num           int
    discountPrice int
}
​
func main() {
    var n int
    fmt.Scan(&n)
​
    order := make(map[int][2]int) // 低价商品 id -> [num, price]
    var res []Product
    reader := bufio.NewReader(os.Stdin)
​
    for i := 0; i < n; i++ {
        var id, num, price, status int
        fmt.Fscan(reader, &id, &num, &price, &status)
        if status != 0 {
            continue // 只处理审批通过
        }
        if price > 100 {
            res = append(res, Product{id, price, num, price})
        } else {
            val := order[id]
            val[0] += num
            val[1] = price
            order[id] = val
        }
    }
​
    // 处理低价商品
    for id, v := range order {
        num, price := v[0], v[1]
        tmp := Product{id, price, num, price}
        if num >= 100 {
            tmp.discountPrice = int(math.Ceil(float64(price) * 0.9))
        }
        res = append(res, tmp)
    }
​
    // 排序
    sort.Slice(res, func(i, j int) bool {
        if res[i].id == res[j].id {
            return res[i].num > res[j].num
        }
        return res[i].id < res[j].id
    })
​
    // 输出
    for _, p := range res {
        fmt.Printf("%d %d %d\n", p.id, p.num, p.discountPrice)
    }
}
