题目描述

在星球争霸篮球赛对抗赛中，最大的宇宙战队希望每个人都能拿到MVP，MVP的条件是单场最高分得分获得者，可以并列。所以宇宙战队决定在比赛中尽可能让更多队员上场，并且让所有得分的选手得分都相同， 然而比赛过程中的每1分钟的得分都只能由某一个人包揽。

输入描述

输入第一行为一个数字 t ，表示为有得分的分钟数 1 ≤ t ≤ 50。

第二行为 t 个数字，代表每一分钟的得分 p， 1 ≤ p ≤ 50。

输出描述

输出有得分的队员都是MVP时，最少得MVP得分。

用例1

输入

xxxxxxxxxx
9 
5 2 1 5 2 1 5 2 1

输出

xxxxxxxxxx
6

说明

样例解释 一共 4 人得分，分别都是 6 分 5 + 1 ， 5 + 1 ， 5 + 1 ， 2 + 2 + 2

 

题解

思路：递归回溯

  1. 暴力枚举允许获得MVP人数可能的情况,可能的情况为[1, t].

  2. 根据1的枚举的MVP人数 x ，判断是否可以满足 指定人数的MVP情况，不满足则将x - 1，继续进行判断，判断过程逻辑如下：

    1. 首先判断sum(score) % x == 0, 不满足直接返回false.
    2. 判断max(score) > (sum(score) / x), 每一场分数不能进行拆分，肯定不满足所有人都拿sum(score) /x的分数。
    3. 使用递归回溯尝试不同组合方案，判断是否能组合出 x 个 分数为 sum(score) / x的方案。其中涉及部门剪枝优化，具体逻辑可参照代码注释。

  3. 执行2的逻辑，解决条件为x == 1 或者 在 [1,t]中可以找到一个合法情况。输出最终的x就是答案。

bool backtrack(int index, int right, vector<int> & buckets, int subSum, vector<int> ans) {
    if (index == right) {
        for (int i = 0; i < buckets.size(); i++) {
            if (buckets[i] != subSum) {
                return false;
            }
        }
        return true;
    }
​
    for (int i = 0; i < buckets.size(); i++) {
         // 剪枝 防止重复运算
        if (i > 0 && buckets[i] == buckets[i - 1]) {
            continue;
        }
       
        if (buckets[i] + ans[index] > subSum) {
            continue;
        }
        // 递归
        buckets[i] += ans[index];
        if (backtrack(index + 1, right, buckets, subSum, ans)) {
            return true;
        }
        // 回溯
        buckets[i] -= ans[index];
        
    }
    return false;
}

bool canPartition(vector<int> ans, int n, int sum) {
    //不能整除说明当前假设mvp人数下，肯定不成立
    if (sum % n) {
        return false;
    }
    // 为每个人得到的分数
    int subSum = sum / n;
    
    int size = ans.size();
    // 分数不能进行分割，所以肯定不成立
    if (ans[0] > subSum) {
        return false;
    }
  
    int beginIndex = 0;
    // 预处理数据，减少数据量，提高算法执行效率
    while (beginIndex < size && ans[beginIndex] == subSum) {
        n--;
        beginIndex++;
    }
​
    if (n == 0) {
        return true;
    }         
​
    vector<int> buckets(n, 0);
    return backtrack(beginIndex, size,buckets, subSum, ans);
}

    int sum = 0;

        sum += ans[i];
    }
    // 将分数进行逆序排序
    sort(ans.begin(), ans.end(), cmp);
    // 从大到小枚举获取MVP的人数， 大小范围为 1 - m
    while (n != 1) {
        if (canPartition(ans, n, sum)) {
            break;
        }
        n--;
    }

}

def backtrack(index, right, buckets, sub_sum, ans, memo):
    if index == right:
        # 所有桶的值都已经达到 sub_sum
        return all(bucket == sub_sum for bucket in buckets)
    
    # 使用 memo 来缓存已经尝试过的状态
    state = tuple(buckets)
    if state in memo:
        return False
    memo.add(state)
    
    for i in range(len(buckets)):
        # 剪枝：避免重复运算
        if i > 0 and buckets[i] == buckets[i - 1]:
            continue
        
        if buckets[i] + ans[index] > sub_sum:
            continue
        
        # 递归
        buckets[i] += ans[index]
        if backtrack(index + 1, right, buckets, sub_sum, ans, memo):
            return True
        # 回溯
        buckets[i] -= ans[index]
    
    return False

def can_partition(ans, n, total_sum):
    if total_sum % n != 0:
        return False
    
    sub_sum = total_sum // n
    if ans[0] > sub_sum:
        return False
​
    begin_index = 0
    while begin_index < len(ans) and ans[begin_index] == sub_sum:
        n -= 1
        begin_index += 1
​
    if n == 0:
        return True
    
    buckets = [0] * n
    memo = set()  # 用于缓存已经遍历的桶状态
    return backtrack(begin_index, len(ans), buckets, sub_sum, ans, memo)

function findMaxAverage(arr, n) {
  // 计算总和并从大到小排序
  const sum = arr.sort((a, b) => b - a).reduce((acc, val) => acc + val, 0);
  
  let count = n;
  // 从n开始枚举MVP人数
  while (count >= 1) {
    if (isPartitionPossible([...arr], sum, count)) {
      return sum / count;
    }
    count--;
  }
}
​
function isPartitionPossible(arr, sum, count) {
  // 如果总和不能均分
  if (sum % count !== 0) return false;
​
  const subSum = sum / count;
  // 如果最大的分数超过每个人应该得到的分数，直接返回false
  if (subSum < arr[0]) return false;
​
  // 剔除已经等于subSum的值，减少数据量
  while (arr[0] === subSum) {
    arr.shift();
    count--;
  }
​
  const buckets = new Array(count).fill(0);
  return canPartitionRecursively(0, arr, subSum, buckets);
}
​
function canPartitionRecursively(index, arr, subSum, buckets) {
  if (index === arr.length) {
    // 所有元素都已经分配完
    return true;
  }
​
  const current = arr[index];
​
  for (let i = 0; i < buckets.length; i++) {
    // 剪枝：避免重复运算
    if (i > 0 && buckets[i] === buckets[i - 1]) continue;
    
    // 如果当前桶加入当前元素没有超过目标分数
    if (buckets[i] + current <= subSum) {
      buckets[i] += current;
      if (canPartitionRecursively(index + 1, arr, subSum, buckets)) return true;
      // 回溯
      buckets[i] -= current;
    }
  }
​
  return false;
}

package main
​
import (
    "fmt"
    "sort"
)
​
func cmp(x, y int) bool {
    return x > y
}
​
func backtrack(index, right int, buckets []int, subSum int, ans []int) bool {
    if index == right {
        for _, bucket := range buckets {
            if bucket != subSum {
                return false
            }
        }
        return true
    }
​
    for i := 0; i < len(buckets); i++ {
        // 剪枝，避免重复运算
        if i > 0 && buckets[i] == buckets[i-1] {
            continue
        }
​
        if buckets[i]+ans[index] > subSum {
            continue
        }
​
        // 递归
        buckets[i] += ans[index]
        if backtrack(index+1, right, buckets, subSum, ans) {
            return true
        }
        // 回溯
        buckets[i] -= ans[index]
    }
​
    return false
}
​
func canPartition(ans []int, n, totalSum int) bool {
    // 如果不能整除，直接返回 false
    if totalSum%n != 0 {
        return false
    }
​
    // 每个人应该得到的分数
    subSum := totalSum / n
    size := len(ans)
​
    // 如果最大的得分大于 subSum，直接返回 false
    if ans[0] > subSum {
        return false
    }
​
    beginIndex := 0
​
    // 预处理数据，减少计算量
    for beginIndex < size && ans[beginIndex] == subSum {
        n--
        beginIndex++
    }
​
    if n == 0 {
        return true
    }
​
    buckets := make([]int, n)
    return backtrack(beginIndex, size, buckets, subSum, ans)
}

func main() {
    var n int
    fmt.Scan(&n) // 输入人数
    ans := make([]int, n)
    totalSum := 0
​
    // 输入所有的得分并计算总分
    for i := 0; i < n; i++ {
        fmt.Scan(&ans[i])
        totalSum += ans[i]
    }
​
    // 将得分按降序排序
    sort.Slice(ans, func(i, j int) bool {
        return cmp(ans[i], ans[j])
    })
​
    // 从大到小枚举获取 MVP 的人数
    for n != 1 {
        if canPartition(ans, n, totalSum) {
            break
        }
        n--
    }
​
    // 输出每个人的分数
    fmt.Println(totalSum / n)
}
