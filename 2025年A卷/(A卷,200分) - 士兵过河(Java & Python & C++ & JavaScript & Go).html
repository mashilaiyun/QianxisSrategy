题目描述

一支N个士兵的军队正在趁夜色逃亡，途中遇到一条湍急的大河。敌军在T的时长后到达河面，没到过对岸的士兵都会被消灭。现在军队只找到了1只小船，这船最多能同时坐上2个士兵。

  1. 当1个士兵划船过河，用时为 a[i]；0 <= i < N
  2. 当2个士兵坐船同时划船过河时，用时为max(a[j],a[i])两士兵中用时最长的。
  3. 当2个士兵坐船1个士兵划船时，用时为 a[i]*10；a[i]为划船士兵用时。
  4. 如果士兵下河游泳，则会被湍急水流直接带走，算作死亡。

请帮忙给出一种解决方案，保证存活的士兵最多，且过河用时最短。

备注

  • 两个士兵的同时划船时，如果划速不同则会导致船原地转圈圈；所以为保持两个士兵划速相同，则需要向划的慢的士兵看齐。
  • 两个士兵坐船时，重量增加吃水加深，水的阻力增大；同样的力量划船速度会变慢；
  • 由于河水湍急大量的力用来抵消水流的阻力，所以2）中过河用时不是a[i] *2，而是a[i] * 10。

输入描述

第一行：N 表示士兵数(0<N<1,000,000) 第二行：T 表示敌军到达时长(0 < T < 100,000,000) 第三行：a[0] a[1] … a[i]… a[N- 1] a[i]表示每个士兵的过河时长。(10 < a[i]< 100; 0<= i< N）

输出描述

第一行：”最多存活士兵数” “最短用时”

题解

思路：动态规划 + 贪心

  • 题目要求尽可能保存士兵最多并且过河时间最短，很容易得出应该优先护送过河时间士兵先走，所以先对数组进行排序。排序过后，过河时间短的士兵无法安全过河，那么过河时间的长的士兵也一定无法过河。

  • 定义dp[]数组，其中dp[i]表示 士兵0 1 .... i士兵安全过河所花费的时间。接下来考虑动态规划的状态转移方程。由于小船容量固定为2，所以dp[i]的状态之和dp[i-1] 和 dp[i-2]相关。基于贪心的思想可以得出以下关系

    • a = dp[i-1] + ans[0] + getMinTime(ans[0], ans[i]): 前 i -1 序号的士兵安全过河的时间 + 0 号士兵将船送回去的时间 + 0和i号士兵过河的时间。
    • b = dp[i-2] + ans[0] + getMinTime(ans[i-1], ans[i]) + ans[1] + getMinTime(ans[0], ans[i]): 前 i -2 序号的士兵安全过河的时间 + 0 号士兵将船送回去的时间 + i -1 和 i号士兵过河的时间 + 1号送船回去的时间 +0 和 1号士兵过河的时间
    • dp[i] = min(a, b)

  • 递推状态转移方程中，如果出现dp[i] > t时,说明 i号士兵无法安全过程可以终止递推了，输出对应结果 i dp[i-1]

    cin >> n;
    cin >> t;
    vector<int> ans(n);
    for (int i = 0; i < n; i++) {
        cin >> ans[i];
    }
    sort(ans.begin(), ans.end());
    // 不能过河
    if (ans[0] > t) {
        cout << "0 0";
        return 0;
    } 
​
    if (n == 1) {
        cout << "1 " << ans[0];
        return 0;
    } 
    // 记录 0 - i 个 士兵已全部过河的时间
    vector<int> dp(n);
    dp[0] = ans[0];
    dp[1] = getMinTime(ans[0], ans[1]);
    if (dp[1] > t) {
        cout << "1 " << ans[0];
        return 0;
    }
    
    for (int i = 2; i < n; i++) {
        // 0 士兵送船回去 接 i过来所使用时间
        int a1 = dp[i-1] + ans[0] + getMinTime(ans[0], ans[i]);
        // 0 士兵将船送回  i - 1 和 i -2  同时过河 1士兵再将船送回去  0 1 士兵同时过河
        int a2 = dp[i-2] + ans[0] + getMinTime(ans[i-1], ans[i]) + ans[1] + getMinTime(ans[0], ans[1]);
        dp[i] = min(a1, a2);
        if (dp[i] > t) {
            cout << i  << " " << dp[i-1];
            return 0;
        }
    }
    cout << n << " " << dp[n-1];
    return 0;
}

function getMinTime(t1, t2) {
    return Math.min(t2, 10 * t1);
}

const readline = require('readline');
const rl = readline.createInterface({
    input: process.stdin,
    output: process.stdout

​
const inputLines = [];
rl.on('line', function (line) {
    inputLines.push(line.trim());
    if (inputLines.length === 3) {
        const n = parseInt(inputLines[0]);
        const t = parseInt(inputLines[1]);
        const ans = inputLines[2].split(' ').map(Number);
        solve(n, t, ans);
        rl.close();
    }
});

function solve(n, t, ans) {
    ans.sort((a, b) => a - b);
    // 不能过河
    if (ans[0] > t) {
        console.log("0 0");
        return;
    }
​
    if (n === 1) {
        console.log(`1 ${ans[0]}`);
        return;
    }
    // 记录 0 - i 个 士兵已全部过河的时间
    const dp = Array(n).fill(0);
    dp[0] = ans[0];
    dp[1] = getMinTime(ans[0], ans[1]);
​
    if (dp[1] > t) {
        console.log(`1 ${ans[0]}`);
        return;
    }
​
    for (let i = 2; i < n; i++) {
        // 0 士兵送船回去 接 i过来所使用时间
        const a1 = dp[i - 1] + ans[0] + getMinTime(ans[0], ans[i]);
        // 0 士兵将船送回  i - 1 和 i -2  同时过河 1士兵再将船送回去  0 1 士兵同时过河
        const a2 = dp[i - 2] + ans[0] + getMinTime(ans[i - 1], ans[i]) + ans[1] + getMinTime(ans[0], ans[1]);
        dp[i] = Math.min(a1, a2);
        if (dp[i] > t) {
            console.log(`${i} ${dp[i - 1]}`);
            return;
        }
    }
​
    console.log(`${n} ${dp[n - 1]}`);
}

package main

import (
    "bufio"

// 一个划船 两个划船 时间较小值
func getMinTime(t1, t2 int) int {
    if 10*t1 < t2 {
        return 10 * t1
    }
    return t2
}
​
func main() {
    reader := bufio.NewReader(os.Stdin)
​
    // 读取第一行 n
    line1, _ := reader.ReadString('\n')
    n, _ := strconv.Atoi(strings.TrimSpace(line1))
​
    // 读取第二行 t
    line2, _ := reader.ReadString('\n')
    t, _ := strconv.Atoi(strings.TrimSpace(line2))
​
    // 读取第三行 ans
    line3, _ := reader.ReadString('\n')
    fields := strings.Fields(line3)
    ans := make([]int, n)
    for i := 0; i < n; i++ {
        ans[i], _ = strconv.Atoi(fields[i])
    }
​
    // 按照升序排序
    sort.Ints(ans)
​
    if ans[0] > t {
        fmt.Println("0 0")
        return
    }
​
    if n == 1 {
        fmt.Printf("1 %d\n", ans[0])
        return
    }
    // 前 0 - i 士兵全部安全过河的时间
    dp := make([]int, n)
    dp[0] = ans[0]
    dp[1] = getMinTime(ans[0], ans[1])
​
    if dp[1] > t {
        fmt.Printf("1 %d\n", ans[0])
        return
    }
​
    for i := 2; i < n; i++ {
        // 0 士兵送船回去 接 i过来所使用时间
        a1 := dp[i-1] + ans[0] + getMinTime(ans[0], ans[i])
        // 0 士兵将船送回  i - 1 和 i -2  同时过河 1士兵再将船送回去  0 1 士兵同时过河
        a2 := dp[i-2] + ans[0] + getMinTime(ans[i-1], ans[i]) + ans[1] + getMinTime(ans[0], ans[1])
        if a1 < a2 {
            dp[i] = a1
        } else {
            dp[i] = a2
        }
        if dp[i] > t {
            fmt.Printf("%d %d\n", i, dp[i-1])
            return
        }
    }
​
    fmt.Printf("%d %d\n", n, dp[n-1])
}
